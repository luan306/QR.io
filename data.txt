CREATE DATABASE inventory;
USE inventory;

-- Bảng user
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    username VARCHAR(50) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    full_name VARCHAR(100),
    role VARCHAR(20) DEFAULT 'user', -- admin / user
    department_id INT
);
select * from users
-- Bảng bộ phận
CREATE TABLE departments (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL
);

-- Bảng thiết bị
CREATE TABLE devices (
    id SERIAL PRIMARY KEY,
    qr_code VARCHAR(255) UNIQUE NOT NULL,
    name VARCHAR(100),
    location VARCHAR(100),
    department_id INT REFERENCES departments(id),
    created_at TIMESTAMP DEFAULT NOW()
);

-- Bảng lịch sử quét
CREATE TABLE scans (
    id SERIAL PRIMARY KEY,
    user_id INT REFERENCES users(id),
    device_id INT REFERENCES devices(id),
    scanned_at TIMESTAMP DEFAULT NOW()
);

USE inventory;

SET SQL_SAFE_UPDATES = 0;

-- ============================================================
-- BƯỚC 1: Thêm scan_department_id vào scans (nếu chưa có)
-- ============================================================
SET @exist = (
  SELECT COUNT(*) FROM information_schema.COLUMNS
  WHERE TABLE_SCHEMA = 'inventory'
    AND TABLE_NAME   = 'scans'
    AND COLUMN_NAME  = 'scan_department_id'
);
SET @sql = IF(@exist = 0,
  'ALTER TABLE scans ADD COLUMN scan_department_id INT',
  'SELECT "scan_department_id already exists" AS msg'
);
PREPARE stmt FROM @sql; EXECUTE stmt; DEALLOCATE PREPARE stmt;

-- Đồng bộ dữ liệu cũ
UPDATE scans SET scan_department_id = scan_department
WHERE scan_department_id IS NULL AND scan_department IS NOT NULL;

-- FK scan_department_id → departments (chỉ thêm nếu chưa có)
SET @fk = (
  SELECT COUNT(*) FROM information_schema.TABLE_CONSTRAINTS
  WHERE TABLE_SCHEMA    = 'inventory'
    AND TABLE_NAME      = 'scans'
    AND CONSTRAINT_NAME = 'fk_scans_scan_dept'
);
SET @sql2 = IF(@fk = 0,
  'ALTER TABLE scans ADD CONSTRAINT fk_scans_scan_dept FOREIGN KEY (scan_department_id) REFERENCES departments(id)',
  'SELECT "fk_scans_scan_dept already exists" AS msg'
);
PREPARE stmt FROM @sql2; EXECUTE stmt; DEALLOCATE PREPARE stmt;

-- ============================================================
-- BƯỚC 2: Thêm created_at vào audit_sessions (nếu chưa có)
-- ============================================================
SET @exist2 = (
  SELECT COUNT(*) FROM information_schema.COLUMNS
  WHERE TABLE_SCHEMA = 'inventory'
    AND TABLE_NAME   = 'audit_sessions'
    AND COLUMN_NAME  = 'created_at'
);
SET @sql3 = IF(@exist2 = 0,
  'ALTER TABLE audit_sessions ADD COLUMN created_at DATETIME DEFAULT CURRENT_TIMESTAMP',
  'SELECT "created_at already exists" AS msg'
);
PREPARE stmt FROM @sql3; EXECUTE stmt; DEALLOCATE PREPARE stmt;

-- Đồng bộ dữ liệu cũ
UPDATE audit_sessions SET created_at = started_at
WHERE created_at IS NULL AND started_at IS NOT NULL;

-- ============================================================
-- BƯỚC 3: FK scans.session_id → audit_sessions (nếu chưa có)
-- ============================================================
SET @fk2 = (
  SELECT COUNT(*) FROM information_schema.TABLE_CONSTRAINTS
  WHERE TABLE_SCHEMA    = 'inventory'
    AND TABLE_NAME      = 'scans'
    AND CONSTRAINT_NAME = 'fk_scans_session'
);
SET @sql4 = IF(@fk2 = 0,
  'ALTER TABLE scans ADD CONSTRAINT fk_scans_session FOREIGN KEY (session_id) REFERENCES audit_sessions(id)',
  'SELECT "fk_scans_session already exists" AS msg'
);
PREPARE stmt FROM @sql4; EXECUTE stmt; DEALLOCATE PREPARE stmt;

-- ============================================================
-- BƯỚC 4: Đồng bộ devices.department_id từ original_department_id
-- ============================================================
UPDATE devices
