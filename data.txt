-- ─── Bảng đợt kiểm kê ─────────────────────────────────────────
CREATE TABLE IF NOT EXISTS inventory_rounds (
  id           INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
  name         VARCHAR(200) NOT NULL                      COMMENT 'Tên đợt, VD: Kiểm kê Q2-2025',
  description  TEXT         NULL                          COMMENT 'Ghi chú / mô tả đợt',
  status       ENUM('draft','active','closed')
               NOT NULL DEFAULT 'draft'                   COMMENT 'draft=mới tạo, active=đang chạy, closed=đã đóng',
  created_by   INT UNSIGNED NULL                          COMMENT 'ID user tạo đợt',
  started_at   DATETIME     NULL                          COMMENT 'Thời điểm bắt đầu chính thức',
  closed_at    DATETIME     NULL                          COMMENT 'Thời điểm đóng đợt',
  created_at   TIMESTAMP    NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at   TIMESTAMP    NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
 
  INDEX idx_rounds_status     (status),
  INDEX idx_rounds_created_by (created_by),
  INDEX idx_rounds_started_at (started_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
 
-- ─── Bảng snapshot thiết bị của từng đợt ─────────────────────
-- Khi tạo đợt mới, copy toàn bộ devices hiện tại vào đây.
-- Đây là dữ liệu độc lập — thay đổi devices sau này không ảnh hưởng.
CREATE TABLE IF NOT EXISTS inventory_round_items (
  id                INT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY,
  round_id          INT UNSIGNED NOT NULL,
 
  -- Snapshot thông tin thiết bị tại thời điểm tạo đợt
  device_id         INT          NOT NULL,
  qr_code           VARCHAR(255) NOT NULL,
  device_name       VARCHAR(200) NULL,
  location          VARCHAR(200) NULL,
  department_id     INT          NULL,
  department_name   VARCHAR(200) NULL,
  device_type_id    INT          NULL,
  device_type_name  VARCHAR(200) NULL,
  section_name      VARCHAR(200) NULL,
  group_name        VARCHAR(200) NULL,
  cost_center_name  VARCHAR(200) NULL,
  floor             TINYINT      NULL,
 
  -- Kết quả kiểm kê
  audited           TINYINT(1)   NOT NULL DEFAULT 0       COMMENT '0=chưa kiểm, 1=đã kiểm',
  audited_by        INT UNSIGNED NULL                     COMMENT 'ID user đã quét',
  audited_by_name   VARCHAR(200) NULL                     COMMENT 'Tên user (cached)',
  audited_at        DATETIME     NULL,
  note              VARCHAR(500) NULL                     COMMENT 'Ghi chú khi quét',
 
  -- Phát hiện dị biệt (thiết bị ở sai phòng, v.v.)
  scanned_dept_id   INT          NULL,
  scanned_dept_name VARCHAR(200) NULL,
  is_mismatch       TINYINT(1)   NOT NULL DEFAULT 0       COMMENT '1 nếu quét tại bộ phận khác dept gốc',
 
  snapshotted_at    TIMESTAMP    NOT NULL DEFAULT CURRENT_TIMESTAMP,
 
  CONSTRAINT fk_round_items_round
    FOREIGN KEY (round_id) REFERENCES inventory_rounds(id)
    ON DELETE CASCADE,
 
  INDEX idx_ri_round    (round_id),
  INDEX idx_ri_device   (device_id),
  INDEX idx_ri_qr       (qr_code),
  INDEX idx_ri_audited  (round_id, audited),
  INDEX idx_ri_dept     (round_id, department_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE OR REPLACE VIEW v_round_summary AS
SELECT
  r.id,
  r.name,
  r.description,
  r.status,
  r.started_at,
  r.closed_at,
  r.created_at,
  u.full_name                                        AS created_by_name,
  COUNT(ri.id)                                       AS total_devices,
  SUM(ri.audited)                                    AS audited_count,
  COUNT(ri.id) - SUM(ri.audited)                     AS pending_count,
  ROUND(SUM(ri.audited) * 100.0 / NULLIF(COUNT(ri.id),0), 1) AS progress_pct
FROM inventory_rounds r
LEFT JOIN inventory_round_items ri ON ri.round_id = r.id
LEFT JOIN users u ON u.id = r.created_by
GROUP BY r.id;
